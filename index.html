        // O / X 상태 변경 함수

        function setStatus(id, value) {

            statusData[id] = value;

            const btnO = document.getElementById(`status-o-${id}`);

            const btnX = document.getElementById(`status-x-${id}`);


            if (value === 'O') {

                btnO.classList.add('active-o');

                btnX.classList.remove('active-x');

            } else {

                btnX.classList.add('active-x');

                btnO.classList.remove('active-o');

            }

        }

        function shareToKakao() {

            const date = document.getElementById('check-date').value;

            const name = document.getElementById('checker-name').value.trim() || '미입력';


            let report = `[동경지구촌교회 새가족팀 물품 점검]\n`;

            report += `📅 일자: ${date}\n`;

            report += `👤 점검자: ${name}\n`;

            report += `-------------------\n`;


            let issues = [];

            items.forEach(item => {

                const status = statusData[item.id] || '-';

                const memo = document.getElementById(`memo-${item.id}`).value.trim();

                

                if (status === 'X' || memo !== '') {

                    report += `• ${item.name}: [${status}] ${memo ? memo : '특이사항 없음'}\n`;

                    issues.push(item.id);

                }

            });


            if (issues.length === 0) {

                report += `✅ 특이사항 없음 (모든 물품 양호)\n`;

            }


            navigator.clipboard.writeText(report).then(() => {

                alert('점검 결과가 복사되었습니다!\n카카오톡 단체방에 붙여넣기(Paste) 해주세요.');

            }).catch(() => {

                alert('복사에 실패했습니다. 권한을 확인해주세요.');

            });

        }
